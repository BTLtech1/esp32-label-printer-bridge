# BTL Print Bridge - Windows setup (finder).
# Run in an Administrator terminal:  irm https://btltech1.github.io/esp32-label-printer-bridge/windows.txt | iex
# Finds BTL Print Bridges on your network, then runs the chosen bridge's own setup script
# (http://<bridge>/windows.ps1), which adds it as a Windows printer.

Write-Host ''
Write-Host 'BTL Print Bridge: add your label printer over Wi-Fi' -ForegroundColor Cyan
$me = [Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()
if (-not $me.IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
  Write-Host 'This needs an Administrator window: right-click Start > Terminal (Admin), then paste the line again.' -ForegroundColor Yellow; return }

Write-Host 'Looking for your bridge on this network (a few seconds)...'
$mine = @(Get-NetIPAddress -AddressFamily IPv4 -ErrorAction SilentlyContinue |
  Where-Object { $_.IPAddress -notlike '127.*' -and $_.IPAddress -notlike '169.254.*' } |
  Select-Object -ExpandProperty IPAddress)
$targets = @()
foreach ($a in $mine) {
  $p = $a.Split('.')
  foreach ($i in 1..254) { $targets += ('{0}.{1}.{2}.{3}' -f $p[0], $p[1], $p[2], $i) }
}
$targets = @($targets | Select-Object -Unique)

# Knock on the raw printing port (9100) of every address at once.
$tries = foreach ($ip in $targets) {
  $c = New-Object Net.Sockets.TcpClient
  [pscustomobject]@{ ip = $ip; c = $c; t = $c.ConnectAsync($ip, 9100) }
}
Start-Sleep -Milliseconds 2000
$open = @()
foreach ($x in $tries) {
  if ($x.t.IsCompleted -and -not $x.t.IsFaulted -and $x.c.Connected) { $open += $x.ip }
  try { $x.c.Close() } catch {}
}

# Ask each one whether it's a BTL Print Bridge.
$bridges = @()
foreach ($ip in $open) {
  try { $sc = [string](Invoke-WebRequest -UseBasicParsing -TimeoutSec 3 -Uri "http://$ip/windows.ps1").Content } catch { continue }
  if ($sc -notmatch 'BTL Print Bridge') { continue }
  $name = $ip
  if ($sc -match "nm = '([^']+)'") { $name = $Matches[1] }
  $printer = ''
  try {
    $st = [string](Invoke-WebRequest -UseBasicParsing -TimeoutSec 3 -Uri "http://$ip/status").Content
    if ($st -match 'Printer:</b>\s*(.*?)<br>') { $printer = ($Matches[1] -replace '<[^>]+>', '' -replace '^connected\s*', '').Trim() }
  } catch {}
  # Already set up on this PC? (a Windows printer pointed at this bridge's IP or name)
  $added = @(Get-Printer -ErrorAction SilentlyContinue | Where-Object { $_.PortName -eq $ip -or $_.PortName -eq $name } | Select-Object -ExpandProperty Name)
  $bridges += [pscustomobject]@{ ip = $ip; name = $name; printer = $printer; added = ($added -join "', '"); script = $sc }
}

if ($bridges.Count -eq 0) {
  Write-Host ''
  Write-Host "Couldn't find a bridge. Check that it's powered, finished setting up, and on the same Wi-Fi as this PC." -ForegroundColor Yellow
  Write-Host "Or find the bridge in your router's list of devices, open its IP address in your browser, and use the Copy button there."
  return
}
$pick = $bridges[0]
if ($bridges.Count -gt 1) {
  Write-Host ''
  Write-Host 'Found these bridges:'
  $def = 0
  for ($i = 0; $i -lt $bridges.Count; $i++) {
    $b = $bridges[$i]
    $what = if ($b.printer) { $b.printer } else { 'no printer plugged in' }
    if ($b.added) { $note = "already added as '" + $b.added + "'" } else { $note = 'NEW'; if (-not $def) { $def = $i + 1 } }
    Write-Host ('  [{0}] {1}   {2}   ({3})' -f ($i + 1), $what, $note, $b.ip) }
  $n = 0
  if ($def) { $ans = Read-Host "Press Enter to add [$def], or type another number" } else { $ans = Read-Host 'All are already added. Type a number to set one up again, or just press Enter to stop' }
  if ([string]::IsNullOrWhiteSpace($ans)) { if (-not $def) { return }; $ans = "$def" }
  if (-not [int]::TryParse("$ans", [ref]$n) -or $n -lt 1 -or $n -gt $bridges.Count) { Write-Host 'That number is not in the list. Paste the line again to retry.' -ForegroundColor Yellow; return }
  $pick = $bridges[$n - 1]
} else {
  $what = if ($pick.printer) { $pick.printer } else { 'no printer plugged in' }
  Write-Host ('Found your bridge ({0}), printer: {1}' -f $pick.ip, $what) -ForegroundColor Green
  if ($pick.added) { Write-Host ("It's already added as '" + $pick.added + "'. Setting it up again (this also fixes stuck print jobs).") }
}
$global:BTLOpened = $false
Invoke-Expression $pick.script

# Older bridges don't offer the desktop shortcut themselves: offer it here.
if (-not $global:BTLOpened) {
  $host2 = $pick.name
  $ok = $false
  try { $t = New-Object Net.Sockets.TcpClient; $ok = $t.ConnectAsync($host2, 80).Wait(3000); $t.Close() } catch {}
  if (-not $ok) { $host2 = $pick.ip }
  $url = "http://$host2/"
  Write-Host ''
  $want = Read-Host "Want a desktop shortcut to the bridge's page (saved fliers, updates)? Type Y and press Enter, or just press Enter to skip"
  if ($want -match '^\s*y') {
    try {
      $desk = [Environment]::GetFolderPath('Desktop')
      $label = if ($pick.printer) { $pick.printer } else { $pick.name -replace '\.local$', '' }
      $lnk = Join-Path $desk ((('BTL Print Bridge - ' + $label) -replace '[\\/:*?"<>|]', '') + '.url')
      Set-Content -Path $lnk -Encoding ASCII -Value "[InternetShortcut]`r`nURL=$url`r`nIconFile=C:\Windows\System32\shell32.dll`r`nIconIndex=16`r`n"
      Write-Host ("Made a desktop shortcut: '" + [IO.Path]::GetFileNameWithoutExtension($lnk) + "'.")
    } catch { Write-Host "Couldn't make the shortcut. You can open $url in your browser instead." -ForegroundColor Yellow }
  }
  Write-Host "Opening the bridge's page: $url"
  Start-Process explorer.exe $url   # explorer opens it in your normal (not Administrator) browser
}
