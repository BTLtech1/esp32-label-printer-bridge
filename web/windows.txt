# BTL Print Bridge - Windows setup (finder).
# Run in an Administrator terminal:  irm https://btltech1.github.io/esp32-label-printer-bridge/windows.txt | iex
# Finds BTL Print Bridges on your network, then runs the chosen bridge's own setup script
# (http://<bridge>/windows.ps1), which adds it as a Windows printer.

Write-Host ''
Write-Host 'BTL Print Bridge: add your label printer over Wi-Fi' -ForegroundColor Cyan
$me = [Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()
if (-not $me.IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
  Write-Host 'This needs an Administrator window: right-click Start > Terminal (Admin), then paste the line again.' -ForegroundColor Yellow; return }

Write-Host 'Looking for your bridge on this network (a few seconds)...'
$mine = @(Get-NetIPAddress -AddressFamily IPv4 -ErrorAction SilentlyContinue |
  Where-Object { $_.IPAddress -notlike '127.*' -and $_.IPAddress -notlike '169.254.*' } |
  Select-Object -ExpandProperty IPAddress)
$targets = @()
foreach ($a in $mine) {
  $p = $a.Split('.')
  foreach ($i in 1..254) { $targets += ('{0}.{1}.{2}.{3}' -f $p[0], $p[1], $p[2], $i) }
}
$targets = @($targets | Select-Object -Unique)

# Knock on the raw printing port (9100) of every address at once.
$tries = foreach ($ip in $targets) {
  $c = New-Object Net.Sockets.TcpClient
  [pscustomobject]@{ ip = $ip; c = $c; t = $c.ConnectAsync($ip, 9100) }
}
Start-Sleep -Milliseconds 2000
$open = @()
foreach ($x in $tries) {
  if ($x.t.IsCompleted -and -not $x.t.IsFaulted -and $x.c.Connected) { $open += $x.ip }
  try { $x.c.Close() } catch {}
}

# Ask each one whether it's a BTL Print Bridge.
$bridges = @()
foreach ($ip in $open) {
  try { $sc = [string](Invoke-WebRequest -UseBasicParsing -TimeoutSec 3 -Uri "http://$ip/windows.ps1").Content } catch { continue }
  if ($sc -notmatch 'BTL Print Bridge') { continue }
  $name = $ip
  if ($sc -match "nm = '([^']+)'") { $name = $Matches[1] }
  $printer = ''
  try {
    $st = [string](Invoke-WebRequest -UseBasicParsing -TimeoutSec 3 -Uri "http://$ip/status").Content
    if ($st -match 'Printer:</b>\s*(.*?)<br>') { $printer = ($Matches[1] -replace '<[^>]+>', '').Trim() }
  } catch {}
  $bridges += [pscustomobject]@{ ip = $ip; name = $name; printer = $printer; script = $sc }
}

if ($bridges.Count -eq 0) {
  Write-Host ''
  Write-Host "Couldn't find a bridge. Check that it's powered, finished setting up, and on the same Wi-Fi as this PC." -ForegroundColor Yellow
  Write-Host "Or open the bridge's page (its address is shown after the Wi-Fi step) and use the Copy button there."
  return
}
$pick = $bridges[0]
if ($bridges.Count -gt 1) {
  Write-Host ''
  Write-Host 'Found more than one bridge:'
  for ($i = 0; $i -lt $bridges.Count; $i++) {
    Write-Host ('  [{0}] {1}  ({2})  printer: {3}' -f ($i + 1), $bridges[$i].name, $bridges[$i].ip, $bridges[$i].printer) }
  $n = 0
  $ans = Read-Host 'Type the number of the one to add and press Enter'
  if (-not [int]::TryParse("$ans", [ref]$n) -or $n -lt 1 -or $n -gt $bridges.Count) { Write-Host 'That number is not in the list. Paste the line again to retry.' -ForegroundColor Yellow; return }
  $pick = $bridges[$n - 1]
} else {
  Write-Host ('Found {0} ({1}), printer: {2}' -f $pick.name, $pick.ip, $pick.printer) -ForegroundColor Green
}
Invoke-Expression $pick.script
